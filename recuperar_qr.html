import { createClient } from "npm:@supabase/supabase-js@2"; const headers = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", "Content-Type": "application/json", }; const reply = (body: Record, status = 200) => new Response(JSON.stringify(body), { status, headers }); async function hash(s: string) { const b = await crypto.subtle.digest( "SHA-256", new TextEncoder().encode(s) ); return Array.from(new Uint8Array(b), x => x.toString(16).padStart(2, "0") ).join(""); } Deno.serve(async (req) => { if (req.method === "OPTIONS") { return new Response("ok", { headers }); } if (req.method !== "POST") { return reply({ ok: false, message: "Método não permitido." }, 405); } const url = Deno.env.get("SUPABASE_URL"); const key = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY"); if (!url || !key) { return reply({ ok: false, message: "Serviço de recuperação não configurado." }, 500); } let body: { cpf?: unknown }; try { body = await req.json(); } catch { return reply({ ok: false, message: "Solicitação inválida." }, 400); } const cpf = String(body.cpf ?? "").replace(/\D/g, ""); if (cpf.length < 9 || cpf.length > 11) { return reply({ ok: false, message: "Informe um CPF válido." }, 400); } const db = createClient(url, key, { auth: { persistSession: false } }); const ip = (req.headers.get("x-forwarded-for") || "unknown") .split(",")[0] .trim(); const ipHash = await hash(ip); const cpfHash = await hash(cpf); const since = new Date(Date.now() - 15 * 60 * 1000).toISOString(); const [a, b] = await Promise.all([ db.from("qr_recovery_attempts") .select("id", { count: "exact", head: true }) .eq("ip_hash", ipHash) .gte("attempted_at", since), db.from("qr_recovery_attempts") .select("id", { count: "exact", head: true }) .eq("cpf_hash", cpfHash) .gte("attempted_at", since) ]); if (a.error || b.error) { return reply({ ok: false, message: "Serviço temporariamente indisponível." }, 503); } if ((a.count ?? 0) >= 5 || (b.count ?? 0) >= 5) { return reply({ ok: false, message: "Muitas tentativas. Aguarde 15 minutos ou procure a equipe do evento." }, 429); } const { error: logError } = await db .from("qr_recovery_attempts") .insert({ ip_hash: ipHash, cpf_hash: cpfHash }); if (logError) { return reply({ ok: false, message: "Serviço temporariamente indisponível." }, 503); } const { data, error } = await db .from("participants") .select("name, qr_token") .eq("cpf", cpf) .maybeSingle(); if (error || !data?.qr_token) { return reply({ ok: false, message: "Cadastro não localizado. Confira o CPF ou procure a equipe do evento." }, 404); } return reply({ ok: true, name: data.name, qr_token: data.qr_token }); });
